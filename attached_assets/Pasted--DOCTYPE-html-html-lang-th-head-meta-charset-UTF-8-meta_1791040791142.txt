    <script>
        var tag = document.createElement('script');
        tag.src = "https://youtube.com";
        var firstScriptTag = document.getElementsByTagName('script')[0];
        firstScriptTag.parentNode.insertBefore(tag, firstScriptTag);

        var player;
        function onYouTubeIframeAPIReady() {
            player = new YT.Player('yt-player');
        }

        const musicToggle = document.getElementById('music-toggle');
        let isPlaying = false;

        function toggleMusic() {
            if (isPlaying) {
                player.pauseVideo();
                musicToggle.innerText = "🔇";
            } else {
                player.playVideo();
                musicToggle.innerText = "🔊";
            }
            isPlaying = !isPlaying;
        }

        function goToStep2() {
            if(player && typeof player.playVideo === 'function') {
                player.playVideo();
                isPlaying = true;
                musicToggle.innerText = "🔊";
            }
            musicToggle.classList.remove('hidden');

            confetti({ particleCount: 100, spread: 70, origin: { y: 0.6 } });

            const s1 = document.getElementById('step-1');
            const s2 = document.getElementById('step-2');

            s1.classList.add('scale-95', 'opacity-0');
            setTimeout(() => {
                s1.classList.add('hidden');
                s2.classList.remove('hidden');
                setTimeout(() => {
                    s2.classList.remove('scale-95', 'opacity-0');
                    s2.classList.add('scale-100', 'opacity-100');
                }, 50);
            }, 300);
        }

        function goToStep3() {
            var end = Date.now() + (1 * 1000);
            (function frame() {
                confetti({ particleCount: 5, angle: 60, spread: 55, origin: { x: 0 } });
                confetti({ particleCount: 5, angle: 120, spread: 55, origin: { x: 1 } });
                if (Date.now() < end) { requestAnimationFrame(frame); }
            }());

            const s2 = document.getElementById('step-2');
            const s3 = document.getElementById('step-3');

            s2.classList.add('scale-95', 'opacity-0');
            setTimeout(() => {
                s2.classList.add('hidden');
                s3.classList.remove('hidden');
                setTimeout(() => {
                    s3.classList.remove('scale-95', 'opacity-0');
                    s3.classList.add('scale-100', 'opacity-100');
                }, 50);
            }, 300);
        }

        function restart() {
            if(player && typeof player.stopVideo === 'function') {
                player.stopVideo();
                isPlaying = false;
            }
            musicToggle.classList.add('hidden');

            const s1 = document.getElementById('step-1');
            const s3 = document.getElementById('step-3');

            s3.classList.add('scale-95', 'opacity-0');
            setTimeout(() => {
                s3.classList.add('hidden');
                s1.classList.remove('hidden');
                setTimeout(() => {
                    s1.classList.remove('scale-95', 'opacity-0');
                    s1.classList.add('scale-100', 'opacity-100');
                }, 50);
            }, 300);
        }
    </script>
